function toggleMortgage(i){const p=currentPlayer(),s=BOARD[i];if(state.owners[i]!==p.id)return;const value=Math.floor(s.price/2);
  if(!state.mortgages[i]){
    if(s.group){const inds=BOARD.map((x,k)=>x.group===s.group?k:-1).filter(k=>k>=0);if(inds.some(k=>state.buildings[k]>0))return alert('بع جميع مباني المجموعة قبل الرهن.')}
    state.mortgages[i]=true;p.cash+=value;log(`<b>${p.name}</b> رهن ${s.name} مقابل ${money(value)}.`)
  }else{const cost=Math.ceil(value*1.1);if(p.cash<cost)return alert('الرصيد غير كافٍ لفك الرهن.');p.cash-=cost;state.mortgages[i]=false;log(`<b>${p.name}</b> فك رهن ${s.name} مقابل ${money(cost)}.`)}render()}

function openTrade(){
  const p=currentPlayer(),others=activePlayers().filter(x=>x.id!==p.id);if(!others.length)return;
  state.phase='modal';const m=openModal('تداول بين اللاعبين',body=>{
    body.innerHTML=`<label>التداول مع</label><select id="tradeTarget" class="cash-input">${others.map(o=>`<option value="${o.id}">${o.name}</option>`).join('')}</select><div id="tradeArea"></div>`;
  });
  const renderTrade=()=>{
    const t=state.players[Number(m.querySelector('#tradeTarget').value)],area=m.querySelector('#tradeArea');
    area.innerHTML=`<div class="trade-columns"><div><h3>${p.name} يعرض</h3><input id="cashA" class="cash-input" type="number" min="0" max="${p.cash}" value="0"><div class="trade-list">${tradePropertyRows(p.id,'a')}</div></div><div><h3>${t.name} يعرض</h3><input id="cashB" class="cash-input" type="number" min="0" max="${t.cash}" value="0"><div class="trade-list">${tradePropertyRows(t.id,'b')}</div></div></div><div class="modal-actions"><button id="submitTrade" class="btn primary">تنفيذ التداول</button></div>`;
    area.querySelector('#submitTrade').onclick=()=>executeTrade(m,p.id,t.id);
  };
  m.querySelector('#tradeTarget').onchange=renderTrade;renderTrade();
}
function tradePropertyRows(pid,prefix){
  return BOARD.map((s,i)=>state.owners[i]===pid?{s,i}:null).filter(Boolean)
    .filter(({s})=>!s.group || BOARD.every((x,k)=>x.group!==s.group || state.buildings[k]===0))
    .map(({s,i})=>`<label class="trade-row"><input type="checkbox" data-trade="${prefix}" value="${i}"><span>${s.name}${state.mortgages[i]?' (مرهون)':''}</span></label>`).join('')||'<p>لا توجد عقارات قابلة للتداول.</p>'
}
function executeTrade(modal,aId,bId){const a=state.players[aId],b=state.players[bId],cashA=Number(modal.querySelector('#cashA').value||0),cashB=Number(modal.querySelector('#cashB').value||0);if(cashA>a.cash||cashB>b.cash)return alert('مبلغ التداول أكبر من الرصيد.');const propsA=[...modal.querySelectorAll('[data-trade="a"]:checked')].map(x=>Number(x.value)),propsB=[...modal.querySelectorAll('[data-trade="b"]:checked')].map(x=>Number(x.value));if(!propsA.length&&!propsB.length&&!cashA&&!cashB)return alert('اختر شيئاً للتداول.');if(!confirm(`${b.name}: هل تقبل هذا التداول؟`))return;a.cash=a.cash-cashA+cashB;b.cash=b.cash-cashB+cashA;propsA.forEach(i=>state.owners[i]=bId);propsB.forEach(i=>state.owners[i]=aId);log(`<b>${a.name}</b> و <b>${b.name}</b> نفذا تداولاً.`);closeModal(modal);state.phase='end';render()}
