async function charge(fromId,amount,toId=null,reason='دفعة'){
  if(amount<=0){state.players[fromId].cash+=Math.abs(amount);log(`<b>${state.players[fromId].name}</b> استلم ${money(Math.abs(amount))} — ${reason}.`);renderPlayers();return true}
  const p=state.players[fromId];
  if(p.cash<amount){
    state.pendingPayment={fromId,amount,toId,reason};
    const solved=await paymentCrisis(p,amount,reason);
    if(!solved)return false;
  }
  p.cash-=amount;if(toId!=null&&!state.players[toId].bankrupt)state.players[toId].cash+=amount;
  log(`<b>${p.name}</b> دفع ${money(amount)}${toId!=null?` إلى <b>${state.players[toId].name}</b>`:' للبنك'} — ${reason}.`);renderPlayers();return true;
}

    },false);
    modal.querySelector('#crisisManage').onclick=()=>{closeModal(modal);openManage(()=>paymentCrisis(player,amount,reason).then(resolve),player.id)};

function declareBankruptcy(playerId,creditorId=null){
  const p=state.players[playerId];p.bankrupt=true;p.cash=0;p.inJail=false;
  BOARD.forEach((s,i)=>{if(state.owners[i]!==playerId)return;
    // Buildings return to bank before property transfer.
    const b=state.buildings[i];if(b===5)state.bankHotels++;else state.bankHouses+=b;state.buildings[i]=0;
    if(creditorId!=null&&!state.players[creditorId].bankrupt){state.owners[i]=creditorId}else{state.owners[i]=null;state.mortgages[i]=false}
  });
  log(`<b>${p.name}</b> أعلن الإفلاس${creditorId!=null?` لصالح ${state.players[creditorId].name}`:' للبنك'}.`);state.phase='end';render();
  if(activePlayers().length<=1)finishGame();
}

async function drawCard(deckName){
  const deck=deckName==='chance'?state.chance:state.community,discard=deckName==='chance'?state.chanceDiscard:state.communityDiscard;
  if(!deck.length){deck.push(...shuffle(discard.splice(0)))}const card=deck.shift();
  await showCard(deckName,card.text);log(`<b>${currentPlayer().name}</b> سحب بطاقة: ${card.text}`);
  if(card.action.kind!=='jailCard')discard.push(card);
  await applyCard(card.action);if(state.phase!=='purchase'&&state.phase!=='modal')state.phase='end';render();
}

async function applyCard(a){const p=currentPlayer();
  if(a.kind==='cash'){if(a.amount>=0){p.cash+=a.amount}else await charge(p.id,-a.amount,null,'بطاقة');return}
  if(a.kind==='move'){await moveTo(a.to,a.collect);await resolveLanding();return}
  if(a.kind==='back'){p.pos=(p.pos-a.spaces+40)%40;render();await resolveLanding();return}
  if(a.kind==='jail'){sendToJail('بطاقة');return}
  if(a.kind==='jailCard'){p.jailCards[a.deck]++;setStatus('احتفظت ببطاقة خروج مجاني من السجن.');return}
  if(a.kind==='repairs'){let total=0;BOARD.forEach((s,i)=>{if(state.owners[i]===p.id){const b=state.buildings[i];if(b===5)total+=a.hotel;else total+=b*a.house}});if(total)await charge(p.id,total,null,'صيانة الممتلكات');return}
  if(a.kind==='each'){
    const others=activePlayers().filter(x=>x.id!==p.id);
    if(a.amount<0){for(const o of others)await charge(p.id,-a.amount,o.id,'بطاقة لكل لاعب')}else{for(const o of others)await charge(o.id,a.amount,p.id,'بطاقة عيد الميلاد')};return
  }
  if(a.kind==='nearestRailroad'){
    const stations=[5,15,25,35];const dest=stations.find(x=>x>p.pos)??5;state.extraRentMultiplier=2;await moveTo(dest,true);await resolveLanding();state.extraRentMultiplier=1;return
  }
  if(a.kind==='nearestUtility'){
    const utils=[12,28];const dest=utils.find(x=>x>p.pos)??12;
    await moveTo(dest,true);
    // Official utility card rule: throw the dice again and pay 10× the new total if owned.
    const d1=1+Math.floor(Math.random()*6),d2=1+Math.floor(Math.random()*6);state.lastRoll=[d1,d2];
    $('#die1').textContent=dieFace(d1);$('#die2').textContent=dieFace(d2);log(`<b>${p.name}</b> أعاد رمي النرد للخدمة: ${d1+d2}.`);
    state.utilitySpecial=true;await resolveLanding();state.utilitySpecial=false;return
  }
}
