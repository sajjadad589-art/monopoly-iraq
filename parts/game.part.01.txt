function buildBoard(){
  const board=$('#board');
  $$('.space').forEach(x=>x.remove());

function gridPosition(i){
  // User-approved orientation:
  // 0 bottom-left (Start), 10 top-left (Jail), 20 top-right (Free Parking), 30 bottom-right (Go To Jail).
  if(i===0)return{r:11,c:1,side:'corner'};
  if(i>0&&i<10)return{r:11-i,c:1,side:'side-left'};
  if(i===10)return{r:1,c:1,side:'corner'};
  if(i>10&&i<20)return{r:1,c:i-9,side:'side-top'};
  if(i===20)return{r:1,c:11,side:'corner'};
  if(i>20&&i<30)return{r:i-19,c:11,side:'side-right'};
  if(i===30)return{r:11,c:11,side:'corner'};
  return{r:11,c:41-i,side:'side-bottom'};
}
function spaceIcon(s){
  if(s.icon)return s.icon;
  if(s.type==='property')return ['🏘️','🏢','🕌','🏙️'][Math.floor(Math.random()*4)];
  if(s.type==='railroad')return '🚂';if(s.type==='utility')return s.name==='الماء'?'🚰':'💡';return '';
}

function tokenPosition(pos,offset=0){
  const o=(offset%3-1)*1.6;
  if(pos===0)return{x:6+o,y:94-o};
  if(pos>0&&pos<10)return{x:6+o,y:94-pos*8.8-o};
  if(pos===10)return{x:6+o,y:6-o};
  if(pos>10&&pos<20)return{x:6+(pos-10)*8.8+o,y:6-o};
  if(pos===20)return{x:94+o,y:6-o};
  if(pos>20&&pos<30)return{x:94+o,y:6+(pos-20)*8.8-o};
  if(pos===30)return{x:94+o,y:94-o};
  return{x:94-(pos-30)*8.8+o,y:94-o};
}

}
